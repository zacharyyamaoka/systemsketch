 106    const inputRef = useRef<HTMLInputElement>(null)

 112    )

 114    const activeItem = visibleMatches[activeIndex]
 115

 117      setInvocation(null)
 118      setQuery('')
 119      setActiveIndex(0)
 120      editor.focus()
 121    }, [editor])
 122
 123    const insert = useCallback((item: ShapeLibraryItem) => {
 124      if (!invocation) return
 125      insertShapeLibraryItemAtPoint(editor, item, invocation.pagePoint)
 126      close()
 127    }, [close, editor, invocation])
 128
 129    useEffect(() => {
 130      const onKeyDown = (event: KeyboardEvent) => {
 131        if (invocation || toolbarSurface || !isPrimitiveSearchKey(event)) return
 132        if (!editor.getIsFocused() || editor.getIsReadonly()) return
 133        if (editor.getEditingShapeId() || editor.inputs.getIsDragging()) return
 134        if (editor.menus.getOpenMenus().length > 0) return
 135        if (isEditableShortcutTarget(event.target) || isEditableShortcutTarget(document.activeElement)) return
 136        if (isChromeShortcutTarget(event.target) || isChromeShortcutTarget(document.activeElement)) return
 137
 138        event.preventDefault()
 139        event.stopImmediatePropagation()
 140        const screenPoint = editor.inputs.getCurrentScreenPoint()
 141        const pagePoint = editor.inputs.getCurrentPagePoint()
 142        setQuery('')
 143        setActiveIndex(0)
 144        setInvocation({
 145          screenPoint: { x: screenPoint.x, y: screenPoint.y },
 146          pagePoint: { x: pagePoint.x, y: pagePoint.y },
 147        })
 148      }
 149      window.addEventListener('keydown', onKeyDown, true)
 150      return () => window.removeEventListener('keydown', onKeyDown, true)
 151    }, [editor, invocation, toolbarSurface])
 152
 153    useEffect(() => {
 154      if (!invocation) return
 155      const focus = () => inputRef.current?.focus({ preventScroll: true })
 156      const dismissOutside = (event: PointerEvent) => {
 157        if (event.target instanceof Node && !rootRef.current?.contains(event.target)) close()
 158      }
 159      const relayout = () => setLayoutRevision((current) => current + 1)
 160      editor.timers.requestAnimationFrame(focus)
 161      window.addEventListener('pointerdown', dismissOutside, true)
 162      window.addEventListener('resize', relayout)
 163      return () => {
 164        window.removeEventListener('pointerdown', dismissOutside, true)
 165        window.removeEventListener('resize', relayout)
 166      }
 167    }, [close, editor, invocation])
 168

 170      setActiveIndex(matches.length > 0 ? 0 : -1)
 171    }, [matches.length, query])
 172
 173    useEffect(() => {

 175      if (!input) return
 176      input.setAttribute('role', 'combobox')
 177      input.setAttribute('aria-autocomplete', 'list')
 178      input.setAttribute('aria-expanded', 'true')
 179      input.setAttribute('aria-controls', listboxId)
 180      if (activeItem) input.setAttribute('aria-activedescendant', `systemsketch-primitive-search-option-${activeItem.id}`)
 181      else input.removeAttribute('aria-activedescendant')
 182    }, [activeItem, listboxId])
 183
 184    const onKeyDownCapture = (event: ReactKeyboardEvent<HTMLDivElement>) => {
 185      if (event.key === 'Escape') {
 186        event.preventDefault()
 187        event.stopPropagation()
 188        close()
 189        return
 190      }
 191      if (event.key === 'ArrowDown' || event.key === 'ArrowUp') {
 192        event.preventDefault()
 193        event.stopPropagation()
 194        setActiveIndex((current) => nextPrimitiveSearchIndex(
 195          current,
 196          event.key === 'ArrowDown' ? 1 : -1,
 197          visibleMatches.length,
 198        ))
 199        return
 200      }
 201      if (event.key === 'Enter') {
 202        event.preventDefault()
 203        event.stopPropagation()
 204        if (activeItem) insert(activeItem)
 205      }
 206    }

 208    if (!invocation) return null

  57    viewport: PrimitiveSearchViewport,
  58    bottomObstacleTop = viewport.h,
  59  ): PrimitiveSearchPlacement {
  60    const safeWidth = Math.max(0, viewport.w - PRIMITIVE_SEARCH_MARGIN * 2)
  61    const safeHeight = Math.max(0, bottomObstacleTop - PRIMITIVE_SEARCH_MARGIN * 2)
  62    const w = Math.min(requested.w, safeWidth)
  63    const h = Math.min(requested.h, safeHeight)
  64
  65    const fitsRight = point.x + PRIMITIVE_SEARCH_GAP + w <= viewport.w - PRIMITIVE_SEARCH_MARGIN
  66    const horizontal = fitsRight ? 'right' : 'left'
  67    const proposedX = horizontal === 'right'
  68      ? point.x + PRIMITIVE_SEARCH_GAP
  69      : point.x - PRIMITIVE_SEARCH_GAP - w
  70
  71    const fitsBelow = point.y + PRIMITIVE_SEARCH_GAP + h <= bottomObstacleTop - PRIMITIVE_SEARCH_MARGIN
  72    const vertical = fitsBelow ? 'below' : 'above'
  73    const proposedY = vertical === 'below'
  74      ? point.y + PRIMITIVE_SEARCH_GAP
  75      : point.y - PRIMITIVE_SEARCH_GAP - h
  76
  77    return {
  78      x: Math.round(clamp(proposedX, PRIMITIVE_SEARCH_MARGIN, viewport.w - PRIMITIVE_SEARCH_MARGIN - w)),
  79      y: Math.round(clamp(proposedY, PRIMITIVE_SEARCH_MARGIN, bottomObstacleTop - PRIMITIVE_SEARCH_MARGIN - h)),
  80      w: Math.round(w),
  81      h: Math.round(h),
  82      horizontal,
  83      vertical,
